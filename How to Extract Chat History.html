How to Export Your Full Chat History

Use this simple guide to safely capture 100% of your chat history directly in your browser.

⚠️ Important Limitations & Disclaimer

  • Text Only: This tool captures text, code snippets, and markdown. It cannot capture or download embedded images, GIFs, videos, or file attachments.
  • Verify Before Deleting: While this tool is highly reliable, network glitches can happen. Always open your downloaded text file and verify that your entire chat was captured successfully BEFORE you delete any chat rooms!

Step 1: Open Your Chat & Dev Console

  1. Navigate to your specific chat room page.
  2. Hit F12 on your keyboard (or use Ctrl+Shift+J on Windows / Cmd+Option+J on Mac).
  3. Select the Console tab at the top of the developer tools panel.

Step 2: Copy & Paste the Interceptor Code

Click the copy button below, paste the script into your browser's console prompt, and press Enter.

(function() {
    window._interceptedMessages = window._interceptedMessages || new Map();

    let oldBadge = document.getElementById('chat-exporter-guide');
    if (oldBadge) oldBadge.remove();

    let guideBox = document.createElement('div');
    guideBox.id = 'chat-exporter-guide';
    guideBox.style.cssText = `
        position: fixed; bottom: 30px; right: 30px; z-index: 999999;
        background: #181818; color: #ffffff; padding: 25px;
        border-radius: 14px; font-family: 'Segoe UI', Tahoma, Geneva, Verdana, sans-serif;
        width: 440px; box-shadow: 0 10px 30px rgba(0,0,0,0.8); border: 3px solid #4ec9b0;
    `;
    
    guideBox.innerHTML = `
        

            📦 Shape Chat Exporter Active
        

        

            Step 1: Scroll all the way up to the very top of your chat history to load older messages.
        

        

            Messages Captured: 0
        

        

            How to use:

            1️⃣ Scroll to the top (watch the count go up)

            2️⃣ Scroll back down to the bottom (grabs recent messages)

            3️⃣ Click the download button below when done!
        

        
    `;
    document.body.appendChild(guideBox);

    function updateUI() {
        const count = window._interceptedMessages.size;
        document.getElementById('exporter-count').innerText = count;
        
        const statusText = document.getElementById('exporter-status-text');
        if (count > 0) {
            statusText.style.borderLeftColor = '#4ec9b0';
            statusText.style.color = '#4ec9b0';
            statusText.innerHTML = `Active: Keep scrolling up until no more history loads, then scroll down to the bottom!`;
        }
    }

    function parseParts(partsArray) {
        if (!Array.isArray(partsArray)) return '';
        return partsArray.map(part => {
            if (typeof part === 'string') return part;
            if (part && typeof part === 'object') {
                return part.text || part.content || part.value || '';
            }
            return '';
        }).join('');
    }

    async function processData(data) {
        let arr = [];
        if (Array.isArray(data)) arr = data;
        else if (Array.isArray(data.messages)) arr = data.messages;
        else if (data.data && Array.isArray(data.data.messages)) arr = data.data.messages;
        else if (data.chat && Array.isArray(data.chat.messages)) arr = data.chat.messages;

        arr.forEach(msg => {
            if (!msg.role) return;
            let text = '';
            let engine = '';

            if (msg.parts && Array.isArray(msg.parts)) {
                text = parseParts(msg.parts);
                let enginePart = msg.parts.find(p => p && typeof p === 'object' && p.ai_engine);
                if (enginePart && enginePart.ai_engine.label) {
                    engine = enginePart.ai_engine.label;
                }
            } else if (msg.partsHistory && msg.selectedVariantIndex !== undefined) {
                text = parseParts(msg.partsHistory[msg.selectedVariantIndex]);
            }

            if (text.trim() !== '') {
                if (!engine) {
                    engine = msg.model || msg.engine || (msg.metadata && msg.metadata.model) || '';
                }
                
                let engineLabel = engine ? ` [${engine}]` : '';
                const senderName = (msg.senderName || (msg.role.toLowerCase() === 'user' ? 'You' : 'AI')) + engineLabel;
                const key = msg.id || `${senderName}:${text.substring(0, 40)}`;
                
                let timestamp = 0;
                if (msg.createdAt) timestamp = new Date(msg.createdAt).getTime();
                else if (msg.timestamp) timestamp = new Date(msg.timestamp).getTime();

                if (!window._interceptedMessages.has(key)) {
                    window._interceptedMessages.set(key, { name: senderName, text: text, timestamp: timestamp });
                }
            }
        });
        updateUI();
    }

    if (!window._originalFetch) {
        window._originalFetch = window.fetch;
        window.fetch = async function(...args) {
            const response = await window._originalFetch.apply(this, args);
            const url = args[0];
            if (typeof url === 'string' && (url.includes('/chat') || url.includes('/messages'))) {
                try {
                    const clone = response.clone();
                    const data = await clone.json();
                    processData(data);
                } catch (e) {}
            }
            return response;
        };
    }

    window.downloadChat = function() {
        let sorted = Array.from(window._interceptedMessages.values());
        sorted.sort((a, b) => a.timestamp - b.timestamp);

        let textOutput = sorted.map(m => {
            let timeString = m.timestamp > 0 ? new Date(m.timestamp).toLocaleString() : "Unknown Time";
            return `[${timeString}] ${m.name}:\n${m.text}\n\n-----------------------------------\n\n`;
        }).join('');
        
        const blob = new Blob([textOutput], { type: 'text/plain;charset=utf-8' });
        const link = document.createElement('a');
        link.href = URL.createObjectURL(blob);
        link.download = 'full_chat_history.txt';
        link.click();

        guideBox.style.borderColor = '#569cd6';
        let statusText = document.getElementById('exporter-status-text');
        statusText.style.borderLeftColor = '#569cd6';
        statusText.style.color = '#569cd6';
        statusText.innerHTML = `Success! Downloaded ${sorted.length} messages.`;

        let btn = document.getElementById('exporter-download-btn');
        btn.style.backgroundColor = '#4ec9b0';
        btn.innerText = '✅ Download Complete!';
    };

    console.log("🟢 Big Guided Interceptor Active with UI Button!");
})();

Step 3: Scroll & Download

  1. Gradually scroll from the bottom of your chat up to the very beginning, waiting 1-2 seconds for older messages to appear.
  2. After reaching the top, scroll back down to the bottom to lock in the newest messages.
  3. Click the 📥 Download Chat File Now button right on your floating screen panel!
  4. Your full_chat_history.txt file will instantly download to your device!
