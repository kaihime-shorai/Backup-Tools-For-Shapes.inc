📚 Interactive Knowledge Base Exporter

Because the Knowledge page uses standard tabs (General, Commands, Relationships, and custom ones), this script launches a clean floating control panel in the corner of your screen. It allows you to freely click through your tabs at your own pace and bundles everything into one master JSON file!

⚠️ HOW IT WORKS:
  • Non-Blocking Widget: A dark control panel will appear in the bottom-right corner of your page when executed.
  • Multi-Tab Support: Go to a tab, click "Capture Tab", switch to your next tab, click capture again, and hit "Download" when finished.

Step 1: Open Your Knowledge Base Page

Navigate to your shape's Knowledge page and make sure you are sitting on your first tab (e.g., General).

Step 2: Open your Developer Console

Press F12 (or right-click anywhere and select Inspect), then click on the Console tab.

Step 3: Run the Floating-UI Script

Click the copy button below, paste the script into your console, and press Enter:

(function() {
    // 1. Find the shape name
    let shapeName = "Shape";
    const nameInput = document.querySelector('input[placeholder="tenshi"]') || document.querySelector('input[value="Mihairi"]');
    if (nameInput && nameInput.value) shapeName = nameInput.value.trim();

    const allTabsData = {};

    // 2. Create a non-blocking floating UI panel
    const panel = document.createElement('div');
    panel.style.cssText = `
        position: fixed;
        bottom: 20px;
        right: 20px;
        background: #161b22;
        color: #c9d1d9;
        border: 1px solid #30363d;
        padding: 15px;
        border-radius: 8px;
        z-index: 999999;
        font-family: -apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, sans-serif;
        box-shadow: 0 8px 24px rgba(0,0,0,0.5);
        width: 300px;
    `;
    
    panel.innerHTML = `
        
📦 Knowledge Exporter

        

            

                
Tabs

                
0

            

            

                
Entries

                
0

            

        

        
Switch to a tab, then click capture.

        

            
            
        

    `;
    
    document.body.appendChild(panel);

    const statusEl = panel.querySelector('#export-status');
    const captureBtn = panel.querySelector('#capture-btn');
    const finishBtn = panel.querySelector('#finish-btn');
    const tabCountEl = panel.querySelector('#tab-count');
    const entryCountEl = panel.querySelector('#entry-count');

    function updateTotals() {
        const tabNames = Object.keys(allTabsData);
        const totalEntries = tabNames.reduce((sum, name) => sum + allTabsData[name].length, 0);
        if (tabCountEl) tabCountEl.textContent = tabNames.length;
        if (entryCountEl) entryCountEl.textContent = totalEntries;
    }

    // 3. Capture button logic
    captureBtn.onclick = () => {
        let activeTabName = "General";
        const activeTab = document.querySelector('.page_tab__StlPv[data-active="true"]');
        if (activeTab) {
            activeTabName = activeTab.textContent.trim().replace(/[^a-zA-Z0-9]/g, "_");
        }

        const cards = document.querySelectorAll('.page_knowledgeCard__QPCDk');
        const scrapedCards = [];

        cards.forEach((card, cardIndex) => {
            const contentEl = card.querySelector('.page_content__BdS5v');
            const metaEl = card.querySelector('.page_meta__chVMO');
            
            scrapedCards.push({
                id: cardIndex + 1,
                content: contentEl ? contentEl.innerText.trim() : "",
                metadata: metaEl ? metaEl.innerText.trim() : ""
            });
        });

        allTabsData[activeTabName] = scrapedCards;
        updateTotals();
        statusEl.innerHTML = \`✅ Captured \${activeTabName} (\${scrapedCards.length} entries). Switch tabs and capture next!\`;
        console.log(\`Captured tab: "\${activeTabName}" with \${scrapedCards.length} items.\`);
    };

    // 4. Finish and download logic
    finishBtn.onclick = () => {
        const exportPayload = {
            Shape: shapeName,
            ExportedAt: new Date().toISOString(),
            Tabs: allTabsData
        };

        const fileName = \`\${shapeName}_Knowledge_Complete_Export.json\`.replace(/\\s+/g, "_");
        const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(exportPayload, null, 2));
        
        const downloadAnchor = document.createElement('a');
        downloadAnchor.setAttribute("href", dataStr);
        downloadAnchor.setAttribute("download", fileName);
        document.body.appendChild(downloadAnchor);
        downloadAnchor.click();
        downloadAnchor.remove();
        
        panel.remove();
        console.log(\`🎉 Knowledge export downloaded as: \${fileName}\`);
    };
})();
